cmake_minimum_required(VERSION 3.21)
project(plmpeg_kernel_tests C)
set(CMAKE_C_STANDARD 99)
enable_testing()
option(PLM_TEST_SANITIZE "Enable native address/undefined sanitizers" ON)
add_executable(plmpeg_idct_test idct_test.c)
add_executable(plmpeg_window_test window_test.c)
foreach(target plmpeg_idct_test plmpeg_window_test)
    target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_LIST_DIR}/..")
    target_compile_options(${target} PRIVATE -O3 -fwrapv -ffp-contract=off)
    if(UNIX AND NOT APPLE)
        target_link_libraries(${target} PRIVATE m)
    endif()
    if(PLM_TEST_SANITIZE)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize=shift)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
endforeach()
add_test(NAME plmpeg_idct COMMAND plmpeg_idct_test)
add_test(NAME plmpeg_window COMMAND plmpeg_window_test)
